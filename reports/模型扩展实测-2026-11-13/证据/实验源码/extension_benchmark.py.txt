def load_plan(task, language, seed):
    path = ORIGINAL_PLANS / (v2.ident(task, language, seed) + ".json")
    assert hashlib.sha256(path.read_bytes()).hexdigest() == PLAN_MANIFEST[path.name]
    return json.loads(path.read_text())


v2.load_plan = load_plan


def load_source(name, path):
    spec = importlib.util.spec_from_loader(name, SourceFileLoader(name, str(path)))
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def post(url, body, timeout=90):
    request = urllib.request.Request(url, data=json.dumps(body).encode(), headers={"Content-Type":"application/json"})
    with urllib.request.urlopen(request, timeout=timeout) as response:
        return json.load(response)


class VerdictModel:
    """Pinned ONNX scoring contract: E5 prefixes, mean pooling, cosine times 20."""
    def __init__(self, path):
        import numpy as np
        import onnxruntime as ort
        from tokenizers import Tokenizer
        self.np = np
        opts = ort.SessionOptions()
        opts.intra_op_num_threads = 2
        opts.inter_op_num_threads = 1
        self.session = ort.InferenceSession(str(path / "onnx/model_quantized.onnx"), opts, providers=["CPUExecutionProvider"])
        self.input_names = {x.name for x in self.session.get_inputs()}
        self.tok = Tokenizer.from_file(str(path / "tokenizer.json"))
        self.tok.enable_truncation(512)
        self.tok.enable_padding(pad_id=self.tok.token_to_id("<pad>") or 0, pad_token="<pad>")
        self.options = {}
        self.calls = 0
        self.tokens = 0

    def encode(self, texts):
        np = self.np
        enc = self.tok.encode_batch(texts)
        ids = np.array([e.ids for e in enc], dtype=np.int64)
        mask = np.array([e.attention_mask for e in enc], dtype=np.int64)
        feed = {"input_ids":ids, "attention_mask":mask}
        if "token_type_ids" in self.input_names:
            feed["token_type_ids"] = np.zeros_like(ids)
        self.calls += 1
        self.tokens += int(mask.sum())
        hidden = self.session.run(None, feed)[0]
        weights = mask[:, :, None].astype(np.float32)
        result = (hidden * weights).sum(1) / np.maximum(weights.sum(1), 1e-9)
        return result / (np.linalg.norm(result, axis=1, keepdims=True) + 1e-9)

    def predict(self, state, questions):
        # Official Jev rendering turns a JSON-object string into readable field lines.
        def render(value, prefix=""):
            if isinstance(value, dict):
                return "\n".join(render(x, f"{prefix}.{k}" if prefix else k) for k, x in value.items())
            if isinstance(value, list):
                return "\n".join(render(x, f"{prefix}[{i}]") for i, x in enumerate(value))
            return f"{prefix}: {value}" if prefix else str(value)
        try:
            parsed = json.loads(state) if isinstance(state, str) else state
        except ValueError:
            parsed = state
        self.calls = self.tokens = 0
        query = self.encode(["query: " + render(parsed)])[0]
        answers = {}
        for qid, question in questions.items():
            criteria = question["criteria"]
            texts = [question["instructions"] + " " + text for text in criteria.values()]
            missing = list(dict.fromkeys(x for x in texts if x not in self.options))
            if missing:
                vectors = self.encode(["passage: " + x for x in missing])
                self.options.update(zip(missing, vectors))
            logits = 20 * (self.np.stack([self.options[x] for x in texts]) @ query)
            probs = self.np.exp(logits-logits.max()); probs /= probs.sum()
            keys = list(criteria)
            answers[qid] = {"choice":keys[int(probs.argmax())], "probabilities":dict(zip(keys, map(float, probs)))}
        return {"answers":answers, "usage":{"backend_forward_calls":self.calls, "input_tokens":self.tokens}, "precision":"INT8"}


class QwenModel:
    def __init__(self, path):
        import psutil
        self.psutil = psutil
        self.url = "http://127.0.0.1:18793"
        binary = ROOT / "vendor/llama/llama-b12360/llama-server"
        self.log = (ROOT / "qwen-server.log").open("w")
        self.process = subprocess.Popen([str(binary), "-m", str(path / "Qwen3.5-0.8B-Q4_0.gguf"),
            "--host", "127.0.0.1", "--port", "18793", "-t", "2", "-tb", "2", "-np", "1",
            "-c", "4096", "-ngl", "0", "--reasoning", "off"], stdout=self.log, stderr=subprocess.STDOUT)
        self.child = psutil.Process(self.process.pid)
        self.peak_rss = 0
        for _ in range(240):
            if self.process.poll() is not None:
                raise RuntimeError("Qwen CPU server exited; see qwen-server.log")
            try:
                with urllib.request.urlopen(self.url + "/health", timeout=1) as r:
                    if json.load(r).get("status") == "ok":
                        self.props = json.load(urllib.request.urlopen(self.url + "/props", timeout=5))
                        (ROOT / "qwen-props.json").write_text(json.dumps(self.props, indent=2))
                        return
            except Exception:
                time.sleep(.25)
        raise RuntimeError("Qwen CPU server readiness timed out")

    def cpu(self):
        t = self.child.cpu_times()
        self.peak_rss = max(self.peak_rss, self.child.memory_info().rss)
        return t.user + t.system

    def predict(self, state, questions):
        answers = {}; receipts = []; cpu0 = self.cpu()
        for qid, q in questions.items():
            keys = list(q["criteria"])
            schema = {"type":"object", "properties":{"choice":{"type":"string","enum":keys}}, "required":["choice"], "additionalProperties":False}
            body = {"model":"qwen3.5-0.8b-q4_0", "temperature":0, "max_tokens":32, "stream":False,
                "cache_prompt":False, "chat_template_kwargs":{"enable_thinking":False},
                "response_format":{"type":"json_schema", "json_schema":{"name":"action", "strict":True, "schema":schema}},
                "messages":[{"role":"system", "content":q["instructions"]+' Return JSON {"choice":"A"} using one offered key.'},
                            {"role":"user", "content":state+"\nOffered choices: "+json.dumps(q["criteria"],ensure_ascii=False)}]}
            raw = post(self.url + "/v1/chat/completions", body)
            message = raw["choices"][0]["message"]
            answer = json.loads(message["content"])
            if answer.get("choice") not in keys:
                raise ValueError("Invalid Qwen closed-choice response")
            answers[qid] = answer
            receipts.append({"qid":qid, "request":body, "response":raw})
        return {"answers":answers, "receipts":receipts, "backend_cpu_ms":(self.cpu()-cpu0)*1000,
                "usage":{"backend_forward_calls":len(questions)}, "precision":"Q4_0", "readout":"constrained JSON generation; thinking disabled"}

    def close(self):
        self.cpu()
        self.process.terminate()
        self.process.wait(timeout=15)
        self.log.close()


class KevModel:
    def __init__(self, path):
        import torch
        sys.path.insert(0, str(ROOT / "vendor/kev-src"))
        from kev.checkpoint import Checkpoint, LoadOptions
        from kev.api import SystemOneRequest, to_record, to_answers
        self.torch = torch
        self.Request = SystemOneRequest
        self.to_record = to_record
        self.to_answers = to_answers
        checkpoint = Checkpoint(str(path))
        # Redirect the pinned base to byte-identical local artifacts, without rewriting checkpoint bytes.
        assert checkpoint.meta.base == "Qwen/Qwen3.5-0.8B-Base"
        assert checkpoint.meta.base_revision == "dc7cdfe2ee4154fa7e30f5b51ca41bfa40174e68"
        checkpoint.meta.base = str(ROOT / "models/qwen-base")
        checkpoint.meta.base_revision = None
        self.tok, self.model = checkpoint.load("cpu", LoadOptions(dtype=torch.float32, backend="torch", attn="eager"))

    def predict(self, state, questions):
        req = self.Request(state=state, questions=questions)
        record, meta = self.to_record(req)
        with self.torch.no_grad():
            enc = self.model.encode(self.tok, record, max_state=2048, max_branch=2048, strict=True)
            logits = self.model.forward(enc)
            probs = [self.torch.softmax(z, -1).cpu().tolist() for z in logits]
        return {"answers":self.to_answers(probs,meta), "usage":{"backend_forward_calls":1,"input_tokens":len(enc["ids"])}, "precision":"FP32", "readout":"official adapter and pointer head"}


def native_page(observed, ordered):
    values = observed["controls"]
    properties = {
        "fill_text":("Text input","textbox",values.get("Text input","")),
        "fill_password":("Password","textbox",values.get("Password","")),
        "fill_area":("Textarea","textbox",values.get("Textarea","")),
        "select_two":("Dropdown (select)","combobox",values.get("Dropdown (select)","")),
        "check_default":("Default checkbox","checkbox",values.get("Default checkbox",False)),
        "radio_default":("Default radio","radio",values.get("Default radio",False)),
        "uncheck_checked":("Checked checkbox","checkbox",values.get("Checked checkbox",False)),
        "fill_revealed":("revealed input","textbox",values.get("revealed_value","")),
        "fill_frame":("iframe1 working","textbox",values.get("iframe1 working","")),
        "fill_main":("main frame working","textbox",""),
        "fill_email":("iframe1 email","textbox",values.get("iframe1 email","")),
    }
    actions = []
    for action in ordered:
        op = v2.OPERATION[action]
        kind = {"TYPE_TEXT":"fill", "SELECT":"select", "CLICK":"click"}.get(op,op.lower())
        entry = {"id":action, "node":action, "kind":kind, "label":v2.base.ACTIONS[action], "role":"button"}
        if action in properties:
            label, role, value = properties[action]
            entry["role"] = role
            entry["value"] = value
            entry["current_value"] = value
            if role in ("checkbox","radio"):
                entry["checked"] = value
            if action == "select_two":
                entry.update(label=label+" → Two", value="2")
        actions.append(entry)
    return {"url":v2.base.ORIGIN+"/"+observed["page"], "title":observed["page"],
            "text":json.dumps(values,ensure_ascii=False), "actions":actions}


class Selector(v2.LocalSelector):
    def __init__(self, name):
        self.name = name
        start = time.perf_counter()
        path = ROOT / "models" / name
        if name.startswith("laya-") or name.startswith("kev-"):
            import torch
            torch.set_num_threads(2)
            torch.set_num_interop_threads(1)
            torch.manual_seed(20261113)
        if name.startswith("laya-"):
            import laya
            self.model = laya.load(str(path), device="cpu", backend="eager")
            self.model.model.encoder.config.reference_compile = False
            if name == "laya-browser":
                self.native = load_source("browser_native", ROOT / "sources/browser-wrapper")
                self.model.cfg["max_len"] = 1024
                self.model.cfg["head_max_len"] = self.model.cfg.get("head_max_len_train",768)
        elif name == "verdict-small":
            self.model = VerdictModel(path)
        elif name == "qwen3.5-0.8b":
            self.model = QwenModel(path)
        elif name == "kev-0.8b":
            self.model = KevModel(path)
        self.load_s = time.perf_counter()-start

    def predict(self, state, ordered, mode, plan_step=None):
        if mode != "native":
            result = super().predict(state,ordered,mode,plan_step)
            response = result["response"]
            result["inference_cpu_ms"] += response.get("backend_cpu_ms",0)
            result["backend_forward_calls"] = response.get("usage",{}).get("backend_forward_calls",result["local_forward_calls"])
            return result
        parsed = json.loads(state)
        page = native_page(parsed["observed"],ordered)
        history = [{"action":v2.base.ACTIONS.get(h["action"],h["action"]),
                    "kind":v2.OPERATION.get(h["action"],""),"page_changed":h["changed"]}
                   for h in parsed["recent_actions"]]
        st, qs, targets, controls = self.native.build_request(page, parsed["goal"], history)
        singles = {qid:next(iter(q["criteria"])) for qid,q in qs.items() if len(q["criteria"])==1}
        submitted = {qid:q for qid,q in qs.items() if qid not in singles}
        start = time.perf_counter(); cpu0=time.process_time()
        answers = self.native.predict_chunked(self.model,st,submitted)
        answers.update({qid:{"choice":key, "deterministic":True} for qid,key in singles.items()})
        op = answers["operation"]["choice"]
        if op in targets:
            selected = targets[op][answers[op.lower()+"_target"]["choice"]]["id"]
        elif op in controls:
            selected = controls[op]["id"]
        elif op in ("DONE","BLOCKED"):
            selected = "__done__" if op=="DONE" else "missing"
        else:
            raise ValueError("Unmapped native operation: "+op)
        return {"selected_action":selected, "operation":op, "response":{"answers":answers}, "questions":qs,
                "criteria":{}, "native_state":st, "native_page":page, "question_count":len(submitted),
                "local_forward_calls":1, "backend_forward_calls":1,
                "deterministic_target":op in targets and len(targets[op])==1,
                "inference_ms":(time.perf_counter()-start)*1000,"inference_cpu_ms":(time.process_time()-cpu0)*1000}


def freeze():
    RESULTS.mkdir(parents=True,exist_ok=True)
    files = [Path(__file__),Path(v2.__file__),Path(v2.project.__file__),Path(v2.base.__file__)]
    files += sorted((ROOT / "sources").glob("browser-*"))
    files += sorted((ROOT / "vendor/kev-src/kev").glob("*.py"))
    files += sorted(ROOT.glob("*-artifacts.json"))
    files += [ROOT / "vendor/llama/llama-b12360/llama-server"]
    protocol = {"version":3,"models":MODELS,"modes":MODES,"browser_extra":"native",
        "dataset":v2.DATASET,"tasks":v2.base.TASKS,"primary":"deepseek-v4.1-flash",
        "plans":"reuse v2 frozen actual primary responses, not new API requests",
        "plan_manifest":PLAN_MANIFEST,"threads":2,"seed":20261113,
        "precisions":{"laya-browser":"FP32","laya-multilingual":"FP32","verdict-small":"ONNX INT8","qwen3.5-0.8b":"GGUF Q4_0","kev-0.8b":"FP32"},
        "qwen":"llama.cpp b12360, 2 threads, 4096 context, no thinking, constrained JSON choice, no response caching",
        "verdict":"512 tokens, E5 query/passage prefixes, mean pooling, cosine*20, option-embedding cache; no query result cache",
        "native":"Author pinned Laya Browser formatter plus host catalog macros/current values, same offered actions; unique-target heads resolved in code; four recent actions",
        "schedule":"One model at a time, alternating history/planned order by episode; native browser mode interleaved",
        "thread_environment":"OMP_NUM_THREADS=2 MKL_NUM_THREADS=2 OPENBLAS_NUM_THREADS=2 TOKENIZERS_PARALLELISM=false; HF_HUB_OFFLINE=1 TRANSFORMERS_OFFLINE=1 for formal runs",
        "limits":"One EN/ZH pair per task, no statistical/general-site claim; fixed host candidates/values/frame/dialog helpers; no production kernel changes; no fallback",
        "inherit":"Same budgets, two-no-progress stop, unchanged independent goal checks; main action_id hidden from local inputs",
        "sources":{str(p):hashlib.sha256(p.read_bytes()).hexdigest() for p in files},
        "environment":{"cpu":"2-core cgroup quota","ram":"8 GiB limit","project_commit":subprocess.check_output(["git","-C",str(v2.project.PROJECT),"rev-parse","HEAD"],text=True).strip()},
        "created_utc":time.strftime("%Y-%m-%dT%H:%M:%SZ",time.gmtime())}
    path = ROOT / "protocol.json"
    assert not path.exists(),"Refusing to overwrite protocol"
    path.write_text(json.dumps(protocol,ensure_ascii=False,sort_keys=True,indent=2))


def protocol_hash():
    path = ROOT / "protocol.json"
    raw=path.read_bytes()
    for source,digest in json.loads(raw)["sources"].items():
        assert hashlib.sha256(Path(source).read_bytes()).hexdigest()==digest,source
    return hashlib.sha256(raw).hexdigest()


def run(name, probe=False):
    RESULTS.mkdir(parents=True,exist_ok=True)
    selector=None
    started=time.perf_counter()
    if name in MODELS:
        try:
            selector=Selector(name)
        except Exception:
            (ROOT/(name+"-load-error.txt")).write_text(traceback.format_exc())
            raise
    elif name=="main":
        selector=v2.MainSelector()
    modes = [*MODES,"native"] if name=="laya-browser" else MODES
    if probe:
        try:
            test=selector.predict('Goal: Fill Text input. Observed: Text input is empty.',
                ["fill_text","fill_area","submit","missing"],"history")
            (ROOT/(name+"-probe.json")).write_text(json.dumps(test,ensure_ascii=False,indent=2))
            print(name,"probe",test["selected_action"],"load_s",selector.load_s,flush=True)
            if name=="laya-browser":
                state=json.dumps({"goal":"Fill Text input then submit.","observed":{"page":"web-form.html",
                    "controls":{"Text input":"","Textarea":""}},"recent_actions":[]})
                native=selector.predict(state,["fill_text","fill_area","submit","missing"],"native")
                (ROOT/(name+"-native-probe.json")).write_text(json.dumps(native,ensure_ascii=False,indent=2))
                print(name,"native probe",native["selected_action"],flush=True)
        finally:
            if hasattr(selector.model,"close"):selector.model.close()
        return
    digest=protocol_hash()
    output=RESULTS/(name+"-episodes.jsonl")
    assert not output.exists(),"Refusing to overwrite trials"
    # Common unscored warm-up; actual outputs retained, no fitting or prompt changes.
    if name in MODELS:
        warm=[]
        for _ in range(2):
            warm.append(selector.predict('Goal: Fill Text input. Observed: Text input is empty.',
                ["fill_text","fill_area","submit","missing"],"history"))
        (ROOT/(name+"-warmup.json")).write_text(json.dumps(warm,ensure_ascii=False,indent=2))
    records=[]
    try:
        with v2.base.sync_playwright() as pw:
            browser=pw.chromium.launch(executable_path="/usr/bin/chromium",headless=True,
                args=["--no-sandbox","--disable-dev-shm-usage","--disable-background-networking"])
            with output.open("w") as out:
                for index,(task,language,seed) in enumerate(v2.DATASET):
                    if name in MODELS:
                        rotation=index%len(modes)
                        current=modes[rotation:]+modes[:rotation]
                    else:
                        current=[{"oracle":"oracle","main":"baseline","plan-direct":"plan_direct"}[name]]
                    for mode in current:
                        result=v2.run_episode(browser,selector,mode,task,language,seed,digest)
                        records.append(result)
                        out.write(json.dumps(result,ensure_ascii=False)+"\n");out.flush()
                        print(json.dumps({k:result.get(k) for k in ["model","mode","task","language","expected_behavior_satisfied","execution_status","reason","e2e_ms"]}),flush=True)
            browser.close()
    finally:
        if selector and hasattr(selector.model if name in MODELS else selector,"close"):
            selector.model.close()
    meta={"model":name,"protocol_sha256":digest,"rounds":len(records),"load_s":selector.load_s if selector else 0,
        "peak_worker_rss_mib":resource.getrusage(resource.RUSAGE_SELF).ru_maxrss/1024,
        "peak_qwen_server_rss_mib":getattr(getattr(selector,"model",None),"peak_rss",0)/1024**2,
        "memory_scope":"Worker process peak includes model load; separate Qwen server sampled RSS; excludes browser/Node/remote service",
        "wall_s":time.perf_counter()-started,
        "versions":{n:importlib.metadata.version(n) for n in ["torch","transformers","laya","onnxruntime","peft","playwright"]}}
    (RESULTS/(name+"-meta.json")).write_text(json.dumps(meta,indent=2))
    print("DONE",name,flush=True)
    if name in ("oracle","plan-direct"):
        assert all(r["expected_behavior_satisfied"] for r in records),"Control failed"


if __name__=="__main__":
    parser=argparse.ArgumentParser()
    parser.add_argument("command",choices=["freeze","oracle","main","plan-direct",*MODELS])
    parser.add_argument("--probe",action="store_true")
    args=parser.parse_args()
    if args.command=="freeze":freeze()
    else:run(args.command,args.probe)
