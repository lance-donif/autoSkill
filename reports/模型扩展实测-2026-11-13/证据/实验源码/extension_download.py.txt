"""Fetch pinned candidate artifacts outside the product checkout."""
import concurrent.futures
import hashlib
import json
from pathlib import Path
import sys
import time
import urllib.request

ROOT = Path(__file__).parent / "extension"
SPECS = {
    "laya-browser": ("cklxx/laya-browser", "browser-api"),
    "laya-multilingual": ("convaiinnovations/laya-multilingual", "multilingual-api"),
    "verdict-small": ("Manav2op/verdict-small", "verdict-api"),
    "qwen3.5-0.8b": ("ggml-org/Qwen3.5-0.8B-GGUF", "qwen-api"),
    "kev-0.8b": ("jaredpalmer/kev-0.8b", "kev-api"),
    "qwen-base": ("Qwen/Qwen3.5-0.8B-Base", "base-api"),
}


def fetch(label):
    repo, metadata = SPECS[label]
    meta = json.loads((ROOT / "sources" / metadata).read_text())
    revision = meta["sha"]
    items = meta["siblings"]
    if label.startswith("laya-"):
        items = [x for x in items if x["rfilename"] in ("rl_agent_config.json", "model.safetensors")
                 or x["rfilename"].startswith(("tokenizer/", "encoder/"))]
    elif label == "verdict-small":
        items = [x for x in items if x["rfilename"] in ("onnx/model_quantized.onnx", "tokenizer.json", "tokenizer_config.json", "config.json")]
    elif label == "qwen3.5-0.8b":
        items = [x for x in items if x["rfilename"] == "Qwen3.5-0.8B-Q4_0.gguf"]
    elif label == "kev-0.8b":
        items = [x for x in items if x["rfilename"] in ("adapter_model.safetensors", "adapter_config.json", "head.pt", "tokenizer.json", "tokenizer_config.json", "training_config.json", "provenance.json")]
    elif label == "qwen-base":
        items = [x for x in items if x["rfilename"] in ("config.json", "model.safetensors-00001-of-00001.safetensors", "model.safetensors.index.json", "tokenizer.json", "tokenizer_config.json")]
    records = []
    for item in items:
        name = item["rfilename"]
        target = ROOT / "models" / label / name
        target.parent.mkdir(parents=True, exist_ok=True)
        start = time.perf_counter()
        if not target.exists():
            temporary = target.with_suffix(target.suffix + ".part")
            with urllib.request.urlopen(f"https://huggingface.co/{repo}/resolve/{revision}/{name}", timeout=120) as src, temporary.open("wb") as dst:
                while chunk := src.read(8 * 1024 * 1024):
                    dst.write(chunk)
            temporary.rename(target)
        assert target.stat().st_size == item["size"], name
        digest = hashlib.file_digest(target.open("rb"), "sha256").hexdigest()
        if item.get("lfs", {}).get("sha256"):
            assert digest == item["lfs"]["sha256"], name
        rec = dict(model=label, repo=repo, revision=revision, file=name, bytes=target.stat().st_size,
                   sha256=digest, download_and_hash_s=time.perf_counter()-start)
        records.append(rec)
        print(json.dumps(rec), flush=True)
    (ROOT / (label + "-artifacts.json")).write_text(json.dumps(records, indent=2))


if __name__ == "__main__":
    with concurrent.futures.ThreadPoolExecutor(3) as pool:
        list(pool.map(fetch, sys.argv[1:]))
