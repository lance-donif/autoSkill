def read(name):
    return [json.loads(line) for line in (RESULTS/(name+'-episodes.jsonl')).read_text().splitlines()]


def stats(rows):
    pos=[r for r in rows if r['task']!='07_missing']
    steps=[s for r in rows for s in r['steps']]
    infer=[s for s in steps if s.get('local_forward_calls',0)>0]
    return dict(n=len(rows),passed=sum(r['expected_behavior_satisfied'] for r in rows),
        blocked=sum(r['execution_status']=='Blocked' for r in rows),
        positive_passed=sum(r['expected_behavior_satisfied'] for r in pos),positive_n=len(pos),
        missing_passed=sum(r['expected_behavior_satisfied'] for r in rows if r['task']=='07_missing'),
        execution_median_s=statistics.median(r['e2e_ms'] for r in rows)/1000,
        execution_total_s=sum(r['e2e_ms'] for r in rows)/1000,
        total_with_plan_s=sum(r['total_with_shared_plan_ms'] for r in rows)/1000,
        total_with_plan_median_s=statistics.median(r['total_with_shared_plan_ms'] for r in rows)/1000,
        shared_planning_s=sum(r['planner_latency_ms'] for r in rows)/1000,
        decisions=len(steps),stored_correct_decisions=sum(s.get('decision_correct',False) for s in steps),
        local_calls=sum(s.get('local_forward_calls',0) for s in steps),
        heads=sum(s.get('question_count',0) for s in infer),
        no_local_call_steps=sum(s.get('local_forward_calls')==0 for s in steps),
        planned_called_steps=sum('matches_main_plan' in s and s.get('local_forward_calls',0)>0 for s in steps),
        planned_called_matches=sum(s.get('matches_main_plan',False) and s.get('local_forward_calls',0)>0 for s in steps),
        inference_median_ms=statistics.median(s['inference_ms'] for s in infer) if infer else None,
        inference_cpu_percent=100*sum(s['inference_cpu_ms'] for s in infer)/sum(s['inference_ms'] for s in infer) if infer else None,
        truncated_calls=sum(bool(s.get('response',{}).get('usage',{}).get('truncated')) or
            bool(s.get('response',{}).get('warnings')) for s in infer),
        recorded_step_api_calls=sum(s.get('api_calls',0) or int(bool(s.get('provider_receipt'))) for s in steps),
        errored_calls=sum(bool(s.get('error')) for s in steps),
        en_passed=sum(r['expected_behavior_satisfied'] for r in rows if r['language']=='en'),
        zh_passed=sum(r['expected_behavior_satisfied'] for r in rows if r['language']=='zh'))


def workbook(rows,summary):

    colors=['FFD966','A9D18E','F4B183']
    def group(name,level):
        index=ws.max_row+1;ws.cell(index,1,name);ws.merge_cells(start_row=index,start_column=1,end_row=index,end_column=12)
        for row in ws.iter_rows(min_row=index,max_row=index,max_col=14):
            for c in row:c.fill=PatternFill('solid',fgColor=colors[level-1]);c.font=Font(name='宋体',bold=True)
        ws.row_dimensions[index].height=23
    group('autoSkill',1);group('浏览器操作',2)
    mapping=[]
    for i,task in enumerate(trial.base.TASKS):
        title=old_report.TITLES[i];group(title,3)
        for number,r in enumerate([r for r in rows if r['task']==task],1):

            steps,expected=old_report.case_text(task)

def main():

    assert len(all_rows)==224

## 四种接法具体改变什么

1. **原始单题**：重新执行上一轮同样的目标＋当前状态、相同问题和候选字母映射，作为本轮配对基线。
2. **单题加历史**：只在原输入后附加最近 4 次真实动作、是否改变观测状态及是否执行报错；候选和问题不变。
3. **操作与目标分层**：用结构化当前状态＋历史，同一 SDK 调用批量询问 operation、goal_progress 和各操作的条件目标；代码选择 operation 对应目标。goal_progress 仅作诊断，不覆盖业务判据。DONE 会被真实结果校验，未成立时记录无效完成和修复提示。此配置同时改变输入结构、问题拆分和动作类别空间，不能把差异完全归因于一个因素。
4. **主模型规划＋局部选择**：DeepSeek 根据初始真实状态和候选一次生成英文原子指令及操作类型；小模型只处理当前指令、当前状态和历史，从该操作的候选中选目标。候选唯一时由代码直接选，记录为零模型推理。原计划 action_id 不进入小模型输入，只供直接回放控制和事后对齐诊断使用。

规划方案同时引入主模型分解、英文归一化、操作类型限定及唯一候选的代码处理，因此提高不能全部归因于小模型。中文轮次中原子指令也是英文，不能声称小模型中文理解能力提高。

“主模型规划后直接回放”使用同一主模型返回的 action_id 顺序。主模型在规划时已经看见完整候选，因此也可能直接解决这些有限任务；这是判定本地选择层是否提供额外价值的必要对照。代码没有用评测正确动作修改或推进小模型的计划，只有实际命令成功派发才推进，最终仍按原需求校验。

## 与完整 Jev-browse 的边界

本轮迁移历史、条件目标、多问题和原子任务分工，没有调用 Jev 云模型，也没有运行官方/社区插件。仍使用原实验的精简观测、预先定义的动作和填写值。候选枚举、通用 DOM 探索、完整观测保留、FOLLOW_UP 优化、Jev 的所有恢复策略和浏览器专项训练均未复现。

为保证与基线可比，各配置保留相同任务步数预算、两次观测不变即停止的执行规则和独立判据。本轮不启用运行时主模型回退，不能把零回退解释为自动化更可靠；规划调用和逐步主模型调用另记。测试预算是实验设置，不是产品功能限制。

执行器复用未经修改的 autoSkill `ui.perform`、`PlaywrightDriver` 原语和 `ui.check`。页面来自冻结的 Selenium 官方 HTML，但浏览器执行器仍是 Playwright；弹窗/缺失报告为宿主辅助操作、iframe 为 Playwright 穿透选择器。不是完整 DSH 工作流或 Selenium WebDriver 验收，也不覆盖生产网页、Windows/UOS、Luna、多语或浏览器专项 checkpoint。候选保留上一轮的固定范围，缺少部分反向恢复动作；不能全面评价自由探索或自主恢复能力。

本轮对前一轮 0/32、3/32、5/32 不做覆盖。新数据用于判断接法的影响，不能混用不同分母或让新版成功删除旧版失败。
