"""Verify immutable protocols, provider traces, report row identities and secret exclusion."""
import hashlib
import json
import os
from pathlib import Path

import retest_v2 as trial


def main():
    digest=trial.frozen_protocol()
    plan_manifest=json.loads((trial.ROOT/'plan-manifest.json').read_text())
    for name,expected in plan_manifest.items():
        assert hashlib.sha256((trial.PLANS/name).read_bytes()).hexdigest()==expected,name
    identities=set();rows=0;controls=0;passed=0;blocked=0
    for path in sorted(trial.RESULTS.glob('*-episodes.jsonl')):
        records=[json.loads(line) for line in path.read_text().splitlines()]
        assert len(records)==(64 if path.name.removesuffix('-episodes.jsonl') in trial.LOCALS else 16),path.name
        for record in records:
            key=tuple(record[k] for k in ['model','mode','task','language','seed'])
            assert key not in identities,key
            identities.add(key)
            assert record['protocol_sha256']==digest,key
            if record['mode']=='oracle':controls+=1
            else:rows+=1
            if record['expected_behavior_satisfied']:
                checks=record['extra'].get('kernel_assertions',[])
                assert checks and all(item['passed'] for item in checks),key
                passed+=record['mode']!='oracle'
            blocked+=record['execution_status']=='Blocked'
            assert abs(record['total_with_shared_plan_ms']-record['e2e_ms']-record['planner_latency_ms'])<0.001,key
            if record['planner_reused']:
                plan=trial.load_plan(record['task'],record['language'],record['seed'])
                assert record['planner_request_key']==(plan.get('receipt') or {}).get('request_key'),key
            for step in record['steps']:
                state=step['input_state']
                assert 'valid_actions' not in state and 'action_id' not in state,key
                assert 'valid_actions' not in json.dumps(step.get('questions',{})),key
                if 'decision_correct' in step:
                    assert step['decision_correct']==(step['selected_action'] in step['valid_actions']),key
                if record['mode']=='planned':
                    actual=json.loads(state)
                    assert actual['goal']==step['atomic_instruction'],key
                    # Only real executed history is permitted, not the planner's hidden ID.
                    assert 'steps' not in actual and 'action_id' not in actual,key
                if record['model'] in trial.LOCALS and step.get('local_forward_calls')==0:
                    assert step.get('question_count')==0,key
            if record.get('screenshot'):
                assert (trial.RESULTS/record['screenshot']).is_file(),key
    assert rows==224 and controls==16,(rows,controls)
    mapping=json.loads((trial.RESULTS/'report-case-mapping.json').read_text())
    assert len(mapping)==224 and len({m['id'] for m in mapping})==224
    for item in mapping:
        assert tuple(item[k] for k in ['model','mode','task','language','seed']) in identities
    requests={};responses={}
    for line in (trial.RESULTS/'api-events.jsonl').read_text().splitlines():
        event=json.loads(line);target=requests if event['event']=='request' else responses
        assert event['request_key'] not in target
        target[event['request_key']]=event
    assert set(requests)==set(responses),'Unpaired provider events'
    cfg_path=Path('/workspace/.autoskills-evaluation-api.json')
    secrets=[json.loads(cfg_path.read_text())['api_key'].encode()] if cfg_path.exists() else []
    if os.environ.get('GH_TOKEN'):secrets.append(os.environ['GH_TOKEN'].encode())
    candidates=[Path(__file__),trial.base.ROOT/'retest_v2.py',trial.base.ROOT/'report_retest_v2.py']
    candidates += [p for p in trial.ROOT.rglob('*') if p.is_file() and p.suffix not in ['.png','.zip','.xlsx']]
    for path in candidates:
        assert not any(s in path.read_bytes() for s in secrets),str(path)
    result={'protocol_sha256':digest,'formal_trials':rows,'controls':controls,'passed':passed,
            'blocked':blocked,'unique_report_rows':224,'immutable_sources_and_plans':True,
            'successful_trials_have_kernel_assertions':True,'no_evaluation_labels_in_provider_inputs':True,
            'paired_api_requests':len(requests),'credential_scan_passed':True,
            'scope':'Trace/readback verification; original browser checks provide semantic verdicts'}
    (trial.RESULTS/'readback-verification.json').write_text(json.dumps(result,indent=2))
    print(json.dumps(result))


if __name__=='__main__':main()
