independent user-goal checks. Provider inputs never contain the evaluation oracle.
"""

    "fill_text":"TYPE_TEXT", "fill_password":"TYPE_TEXT", "fill_area":"TYPE_TEXT",
    "fill_revealed":"TYPE_TEXT", "fill_frame":"TYPE_TEXT", "fill_main":"TYPE_TEXT",

    "accept_confirm":"CLICK", "accept_alert":"CLICK", "wait":"WAIT",

}
OP_DESCRIPTIONS = {
    "TYPE_TEXT":"Enter text in an editable field.",
    "SELECT":"Choose a dropdown option.",
    "CLICK":"Click a button, checkbox, radio or dialog control.",
    "WAIT":"Wait for an expected control to become visible.",
    "RELOAD":"Reload this page.",
    "BLOCKED":"A required element is absent; report it missing.",
    "DONE":"All requested outcomes have already been observed.",
}
OP_QUESTION = "Which operation advances the goal now? Use current values and recent actions; avoid completed steps."
TARGET_QUESTION = "For operation {operation}, which offered target fulfills the current goal? Do not repeat a satisfied field."
PROGRESS_QUESTION = "Does the observed state establish every requested outcome? An action intention is not a completed outcome."
PLAN_SYSTEM = """Plan one bounded browser task from its goal and actual initial observation.
Return only JSON: {"steps":[{"operation":"TYPE_TEXT|SELECT|CLICK|WAIT|RELOAD|BLOCKED","instruction":"one standalone atomic instruction in English","action_id":"one offered action ID"}]}.
Each step must be one concrete operation. Use the goal's exact values. Include every required outcome and order constraint.
Use current values to omit already satisfied work. Report a required element missing when the observation establishes absence.
Do not change unrelated fields or invent requirements. Do not add an unrequested submission.
For the local selector, each instruction must be understandable alone without reading another step.
Never put action IDs or candidate letters into the instruction. The action_id is retained separately for a direct-plan control.
Produce at most 10 steps. Only offered actions may be used. Do not output explanations or assertions of success."""


def ident(task, language, seed):
    return f"{task}-{language}-{seed}"


def append_event(name, data):
    RESULTS.mkdir(parents=True, exist_ok=True)
    with (RESULTS / name).open("a") as out:
        out.write(json.dumps(data, ensure_ascii=False) + "\n")


class RequestFailure(RuntimeError):
    def __init__(self, message, receipt):
        super().__init__(message)
        self.receipt = receipt


class MainAPI:
    def __init__(self):
        self.cfg = json.loads(Path("/workspace/.autoskills-evaluation-api.json").read_text())

    def call(self, system, prompt, purpose, max_tokens):
        body = {"model":self.cfg["model"], "temperature":0, "stream":False,
                "max_tokens":max_tokens, "response_format":{"type":"json_object"},
                "messages":[{"role":"system","content":system},{"role":"user","content":prompt}]}
        request_key = str(uuid.uuid4())
        append_event("api-events.jsonl", {"event":"request","request_key":request_key,
                     "purpose":purpose,"body":body})
        start = time.perf_counter()
        cpu_start = time.process_time()
        raw = None
        try:
            request = urllib.request.Request(self.cfg["base_url"].rstrip("/")+"/v1/chat/completions",
                data=json.dumps(body).encode(),headers={"Content-Type":"application/json",
                "Authorization":"Bearer "+self.cfg["api_key"]})
            with urllib.request.urlopen(request, timeout=90) as response:
                raw = json.load(response)
            choices = raw.get("choices")
            first = choices[0] if isinstance(choices,list) and choices else {}
            message = first.get("message") or {}
            content = message.get("content")
            safe = {"model":raw.get("model"), "request_id":raw.get("id"),"usage":raw.get("usage",{}),
                    "finish_reason":first.get("finish_reason"),"content":content}
            if not isinstance(content,str) or not content.strip():
                raise ValueError("Primary response has no usable message content")
            content = content.strip()
            if content.startswith("```"):
                content = "\n".join(content.splitlines()[1:-1])
            output = json.loads(content)
            if not isinstance(output,dict):
                raise ValueError("Primary response JSON must be an object")
            receipt = {"request_key":request_key,"purpose":purpose,**safe,
                       "latency_ms":(time.perf_counter()-start)*1000,
                       "cpu_ms":(time.process_time()-cpu_start)*1000,"status":"ok"}
            append_event("api-events.jsonl", {"event":"response",**receipt})
            return output, receipt
        except Exception as exc:
            safe = {}
            if isinstance(raw,dict):
                safe = {k:raw.get(k) for k in ["id","model","usage"]}
                safe["choices"] = [{"finish_reason":c.get("finish_reason"),
                    "content":(c.get("message") or {}).get("content")} for c in raw.get("choices",[])]
            receipt = {"request_key":request_key,"purpose":purpose,"status":"error",
                "latency_ms":(time.perf_counter()-start)*1000,"cpu_ms":(time.process_time()-cpu_start)*1000,
                "error":f"{type(exc).__name__}: {exc}","response":safe}
            append_event("api-events.jsonl", {"event":"error",**receipt})
            raise RequestFailure(receipt["error"],receipt) from None


def frozen_protocol():
    raw = (ROOT / "protocol.json").read_bytes()
    obj = json.loads(raw)
    for path, expected in obj["source_hashes"].items():
        assert hashlib.sha256(Path(path).read_bytes()).hexdigest() == expected, path
    return hashlib.sha256(raw).hexdigest()

    files = [Path(__file__),base.ROOT/"benchmark.py",base.ROOT/"project_benchmark.py"]

        "main_model":"deepseek-v4.1-flash","local_models":LOCALS,"dtype":"float32","threads":2,
        "source_hashes":{str(p):hashlib.sha256(p.read_bytes()).hexdigest() for p in files},
        "project_commit":subprocess.check_output(["git","-C",str(project.PROJECT),"rev-parse","HEAD"],text=True).strip(),
        "execution":"Same task budgets and two-unchanged-action stopping as v1; same helpers, candidates and independent verdicts",
        "split_done":"DONE gets independent verification; rejected DONE is recorded as a no-op with a repair hint",
        "split_progress":"Goal-progress head is diagnostic; never overrides an independent business oracle",

        "planned":"Main plans are generated once for each task/language, frozen and reused by all local models and direct-plan control; local input excludes main action_id",

def history_state(goal, observation, history, mode, instruction=None):
    if mode == "baseline":
        return "Goal: " + goal + "\nObserved: " + json.dumps(observation,ensure_ascii=False,separators=(",",":"))
    recent = [{"action":h["selected_action"],"changed":h.get("state_changed"),
               **({"error":True} if h.get("execution_error") else {})} for h in history[-4:]]
    state = {"goal":instruction if mode=="planned" else goal,
             "observed":observation,"recent_actions":recent}
    if history and history[-1]["selected_action"] == "__done__":
        state["repair"] = "Completion was not established. Continue the unfinished goal."
    return json.dumps(state,ensure_ascii=False,separators=(",",":"))


def typed_questions(ordered, mode, plan_step=None):
    choices = {chr(65+i):base.ACTIONS[a] for i,a in enumerate(ordered)}
    if mode in ("baseline","history"):
        return {"next_action":{"type":"choice","instructions":base.INSTRUCTIONS,"criteria":choices}}, {}, choices
    pools = {}
    for key,action in zip(choices,ordered):
        op = OPERATION[action]
        if op not in ("BLOCKED","WAIT","RELOAD"):
            pools.setdefault(op,{})[key] = action
    if mode == "planned":
        op = plan_step["operation"]
        questions = {}
        if op in pools and len(pools[op])>1:
            questions["target"]={"type":"choice","instructions":TARGET_QUESTION.format(operation=op),
                "criteria":{key:choices[key] for key in pools[op]}}
        return questions, pools, choices
    operations = {OPERATION[a]:OP_DESCRIPTIONS[OPERATION[a]] for a in ordered}
    operations["DONE"] = OP_DESCRIPTIONS["DONE"]
    questions = {
        "operation":{"type":"choice","instructions":OP_QUESTION,"criteria":operations},
        "goal_progress":{"type":"choice","instructions":PROGRESS_QUESTION,"criteria":{
            "SATISFIED":"All requested results are observed.",
            "INCOMPLETE":"Required work remains.","UNCERTAIN":"Available observations cannot establish the outcome."}},
    }
    for op,pool in pools.items():
        if len(pool)<2:
            continue
        questions[op.lower()+"_target"]={"type":"choice","instructions":TARGET_QUESTION.format(operation=op),
            "criteria":{key:choices[key] for key in pool}}
    return questions,pools,choices


class LocalSelector:
    def __init__(self, name):
        self.name = name
        self.loaded = base.Decider(name)
        self.model = self.loaded.model
        self.load_s = self.loaded.load_s

    def predict(self, state, ordered, mode, plan_step=None):
        questions,pools,criteria = typed_questions(ordered,mode,plan_step)
        start = time.perf_counter(); cpu_start = time.process_time()
        response = {"answers":{},"usage":{}}
        if questions:
            response = (self.model.system_one(state,questions) if self.name=="opendecider-nano"
                        else self.model.predict(state,questions))
        answers = response["answers"]
        deterministic = False
        if mode in ("baseline","history"):
            key = answers["next_action"]["choice"]
            if key not in criteria:raise ValueError("Invalid next-action key")
            selected = ordered[ord(key)-65]
            op = OPERATION[selected]
        else:
            op = plan_step["operation"] if mode=="planned" else answers["operation"]["choice"]
            if op in ("BLOCKED","WAIT","RELOAD","DONE"):
                selected = {"BLOCKED":"missing","WAIT":"wait","RELOAD":"reload","DONE":"__done__"}[op]
                deterministic = mode=="planned"
            else:
                pool = pools.get(op,{})
                if len(pool)==1:
                    selected = next(iter(pool.values()));deterministic = True
                elif pool:
                    key = answers["target" if mode=="planned" else op.lower()+"_target"]["choice"]
                    if key not in pool:raise ValueError("Invalid target key")
                    selected = pool[key]
                else:
                    raise ValueError("Operation has no offered target")
        return {"selected_action":selected,"response":response,"questions":questions,"criteria":criteria,
                "operation":op,"question_count":len(questions),"local_forward_calls":int(bool(questions)),
                "deterministic_target":deterministic,"inference_ms":(time.perf_counter()-start)*1000,
                "inference_cpu_ms":(time.process_time()-cpu_start)*1000}


class MainSelector:
    name = "deepseek-v4.1-flash"
    load_s = 0
    def __init__(self):self.api=MainAPI()

    def predict(self,state,ordered,mode,plan_step=None):
        choices = {chr(65+i):base.ACTIONS[a] for i,a in enumerate(ordered)}
        answer,receipt = self.api.call(base.INSTRUCTIONS+' Return only JSON {"choice":"A"} using an offered letter.',
            state+"\nAvailable actions: "+json.dumps(choices,ensure_ascii=False),"step-selection",2048)
        key=answer.get("choice")
        if key not in choices:raise RequestFailure("Invalid primary candidate",receipt)
        return {"selected_action":ordered[ord(key)-65],"questions":{},"criteria":choices,
                "response":{"answers":{"next_action":{"choice":key}},"model":receipt["model"],
                "usage":receipt["usage"],"receipt":receipt},"question_count":1,"local_forward_calls":0,
                "api_calls":1,"inference_ms":receipt["latency_ms"],"inference_cpu_ms":receipt["cpu_ms"]}


def plan_path(task,language,seed):return PLANS/(ident(task,language,seed)+".json")


def prepare_plans():
    protocol=frozen_protocol()
    api=MainAPI()
    with base.sync_playwright() as pw:
        browser=pw.chromium.launch(executable_path="/usr/bin/chromium",headless=True,
            args=["--no-sandbox","--disable-dev-shm-usage","--disable-background-networking"])
        for task,language,seed in DATASET:
            path=plan_path(task,language,seed)
            assert not path.exists(), "Refusing to overwrite a plan"
            context=browser.new_context(service_workers="block");context.route("**/*",base.fixture_route)
            page=context.new_page();page.goto(base.ORIGIN+"/"+base.TASKS[task]["page"],wait_until="load")
            observation=base.observe(page,base.TASKS[task],{})
            offered={a:{"description":base.ACTIONS[a],"operation":OPERATION[a]} for a in base.TASKS[task]["actions"]}
            payload={"goal":base.TASKS[task][language],"observation":observation,"offered_actions":offered}
            record={"protocol_sha256":protocol,"task":task,"language":language,"seed":seed,"input":payload}
            try:
                plan,receipt=api.call(PLAN_SYSTEM,json.dumps(payload,ensure_ascii=False),"plan-"+ident(task,language,seed),4096)
                steps=plan.get("steps")
                if not isinstance(steps,list) or not 1<=len(steps)<=10:
                    raise ValueError("Invalid plan step list")
                for step in steps:
                    if (not isinstance(step,dict) or step.get("action_id") not in offered or
                        step.get("operation")!=OPERATION[step["action_id"]] or
                        not isinstance(step.get("instruction"),str) or not step["instruction"].strip()):
                        raise ValueError("Invalid plan step shape or unsupported action")
                record.update(status="ok",steps=steps,receipt=receipt)
            except Exception as exc:
                record.update(status="Blocked",error=f"{type(exc).__name__}: {exc}",receipt=getattr(exc,"receipt",None))
            path.write_text(json.dumps(record,ensure_ascii=False,indent=2))
            print("PLAN",ident(task,language,seed),record["status"],len(record.get("steps",[])),flush=True)
            context.close()
        browser.close()
    (ROOT/"plan-manifest.json").write_text(json.dumps({p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in PLANS.glob("*.json")},indent=2))


def load_plan(task,language,seed):
    path=plan_path(task,language,seed)
    expected=json.loads((ROOT/"plan-manifest.json").read_text())[path.name]
    assert hashlib.sha256(path.read_bytes()).hexdigest()==expected
    return json.loads(path.read_text())


def run_episode(browser, selector, mode, task, language, seed, protocol):
    definition=base.TASKS[task]
    name=selector.name if selector else mode
    ordered=list(definition["actions"]);random.Random(seed*100+int(task[:2])).shuffle(ordered)
    plan=load_plan(task,language,seed) if mode in ("planned","plan_direct") else None
    record={"model":name,"mode":mode,"task":task,"language":language,"seed":seed,
        "protocol_sha256":protocol,"expected_behavior_satisfied":False,"execution_status":"Completed",
        "reason":"budget_exhausted","steps":[],"extra":{},"planner_latency_ms":0,
        "planner_reused":bool(plan),"planner_request_key":None}
    if plan:
        record["planner_latency_ms"]=(plan.get("receipt") or {}).get("latency_ms",0)
        record["planner_request_key"]=(plan.get("receipt") or {}).get("request_key")
        if plan["status"]!="ok":
            record.update(execution_status="Blocked",reason="planner_unavailable",e2e_ms=0,
                          error=plan.get("error"),total_with_shared_plan_ms=record["planner_latency_ms"])
            return record
    context=browser.new_context(viewport={"width":1280,"height":900},service_workers="block")
    context.route("**/*",base.fixture_route)
    page=context.new_page();page.set_default_timeout(1600)
    extra=record["extra"];history=record["steps"]
    start=time.perf_counter();cpu_start=time.process_time();unchanged=0;plan_index=0
    try:
        page.goto(base.ORIGIN+"/"+definition["page"],wait_until="load")
        for index in range(definition["budget"]):
            obs=base.observe(page,definition,extra)
            current=None
            if plan:
                if plan_index>=len(plan["steps"]):
                    record["reason"]="plan_exhausted_before_goal";break
                current=plan["steps"][plan_index]
            state=history_state(definition[language],obs,history,mode,current["instruction"] if current else None)
            # The independent reference is recorded after prediction, never passed to any real provider.
            step={"step":index,"input_state":state,"observation":obs,"plan_index":plan_index if plan else None,
                  "atomic_instruction":current["instruction"] if current else None}
            history.append(step)
            # Construct provider history without the pending record; this avoids a hidden schema change.
            state=history_state(definition[language],obs,history[:-1],mode,current["instruction"] if current else None)
            step["input_state"]=state
            append_event("step-events.jsonl",{"event":"before_decision","model":name,"mode":mode,
                "episode":ident(task,language,seed),"step":index,"input_state":state})
            try:
                if mode=="oracle":
                    valid=base.correct_actions(task,obs,extra)
                    action=next(a for a in ordered if a in valid)
                    prediction={"selected_action":action,"response":{},"questions":{},"question_count":0,
                        "local_forward_calls":0,"inference_ms":0,"inference_cpu_ms":0,"criteria":{}}
                elif mode=="plan_direct":
                    prediction={"selected_action":current["action_id"],"response":{},"questions":{},
                        "question_count":0,"local_forward_calls":0,"inference_ms":0,
                        "inference_cpu_ms":0,"criteria":{},"deterministic_target":True}
                else:
                    prediction=selector.predict(state,ordered,mode,current)
                step.update(prediction)
            except Exception as exc:
                step.update(error=f"{type(exc).__name__}: {exc}",provider_receipt=getattr(exc,"receipt",None))
                record.update(execution_status="Blocked",reason="provider_response_error")
                append_event("step-events.jsonl",{"event":"decision_error","model":name,"mode":mode,
                    "episode":ident(task,language,seed),"step":index,"record":step})
                break
            action=step["selected_action"]
            valid=base.correct_actions(task,obs,extra)
            step.update(valid_actions=valid,decision_correct=action in valid)
            if current:step["matches_main_plan"]=(action==current["action_id"])
            append_event("step-events.jsonl",{"event":"decision","model":name,"mode":mode,
                "episode":ident(task,language,seed),"step":index,"record":step})
            t_action=time.perf_counter();executed=True
            try:
                if action!="__done__":base.execute(page,action,extra)
            except Exception as exc:
                executed=False;step["execution_error"]=f"{type(exc).__name__}: {str(exc).splitlines()[0]}"
            step["execution_ms"]=(time.perf_counter()-t_action)*1000
            after=base.observe(page,definition,extra)
            step.update(after=after,state_changed=(after!=obs))
            terminal,success,reason=base.verdict(task,page,after,extra)
            append_event("step-events.jsonl",{"event":"after_action","model":name,"mode":mode,
                "episode":ident(task,language,seed),"step":index,"record":step})
            if terminal:
                record.update(expected_behavior_satisfied=success,reason=reason);break
            if plan and executed:plan_index+=1
            unchanged=unchanged+1 if after==obs else 0
            if unchanged>=2:
                record["reason"]="two_actions_without_state_progress";break
        record["final_observation"]=base.observe(page,definition,extra)
        record["final_url"]=page.url
    except Exception as exc:
        record.update(execution_status="Blocked",reason="browser_or_adapter_error",error=f"{type(exc).__name__}: {exc}")
    finally:
        record.update(e2e_ms=(time.perf_counter()-start)*1000,host_cpu_ms=(time.process_time()-cpu_start)*1000)
        record["total_with_shared_plan_ms"]=record["e2e_ms"]+record["planner_latency_ms"]
        # Capture evidence outside the measured execution interval; mask synthetic password fields too.
        try:
            directory=RESULTS/"screenshots";directory.mkdir(exist_ok=True)
            image_name=f"{name}-{mode}-{ident(task,language,seed)}.png"
            page.screenshot(path=str(directory/image_name),full_page=True,mask=[page.locator('input[type=password]')],timeout=4000)
            record["screenshot"]="screenshots/"+image_name
        except Exception as exc:
            record["screenshot_error"]=type(exc).__name__
        context.close()
    return record


def run_group(name):
    protocol=frozen_protocol()
    output=RESULTS/(name+"-episodes.jsonl")
    assert not output.exists(), "Refusing to overwrite existing trials"
    selector=None
    if name in LOCALS:selector=LocalSelector(name)
    elif name=="main":selector=MainSelector()
    if name in LOCALS:
        for _ in range(3):selector.loaded.predict('Goal: Fill Text input. Observed: Text input is empty.',
            ["fill_text","fill_area","submit","missing"],[])
    records=[]
    with base.sync_playwright() as pw:
        browser=pw.chromium.launch(executable_path="/usr/bin/chromium",headless=True,
            args=["--no-sandbox","--disable-dev-shm-usage","--disable-background-networking"])
        with output.open("w") as out:
            for episode_index,(task,language,seed) in enumerate(DATASET):
                if name in LOCALS:
                    rotate=episode_index%len(MODES);modes=MODES[rotate:]+MODES[:rotate]
                else:modes=[{"main":"baseline","oracle":"oracle","plan-direct":"plan_direct"}[name]]
                for mode in modes:
                    r=run_episode(browser,selector,mode,task,language,seed,protocol)
                    records.append(r);out.write(json.dumps(r,ensure_ascii=False)+"\n");out.flush()
                    print(json.dumps({k:r.get(k) for k in ["model","mode","task","language",
                        "expected_behavior_satisfied","execution_status","reason","e2e_ms"]}),flush=True)
        browser.close()
    meta={"name":name,"protocol_sha256":protocol,"episodes":len(records),
          "passed":sum(r["expected_behavior_satisfied"] for r in records),
          "blocked":sum(r["execution_status"]=="Blocked" for r in records),
          "peak_host_rss_mib":resource.getrusage(resource.RUSAGE_SELF).ru_maxrss/1024,
          "load_s":selector.load_s if selector else 0,
          "memory_scope":"Whole worker process across modes, including load; excludes Chromium/Node and remote service",
          "versions":{n:importlib.metadata.version(n) for n in ["torch","transformers","opendecider","laya","playwright"]}}
    (RESULTS/(name+"-meta.json")).write_text(json.dumps(meta,indent=2))
    print("DONE",json.dumps(meta),flush=True)
    if name=="oracle" and not all(r["expected_behavior_satisfied"] for r in records):
        raise SystemExit("Executor control failed")


def main():
    parser=argparse.ArgumentParser()
    parser.add_argument("command",choices=["freeze","plans","oracle","main","plan-direct",*LOCALS])
    args=parser.parse_args()
    if args.command=="freeze":print(freeze())
    elif args.command=="plans":prepare_plans()
    else:run_group(args.command)


if __name__=="__main__":
    main()
